# উদাহরণ ফাইল (Examples)

## দ্রুত ডেমো

```python
# চীনা থেকে বাংলায় প্রতিবর্ণীকরণ
import sys
sys.path.insert(0, '../চীনা-বাংলা অনুবাদ')
from transliteration import transliterate_hanzi_to_bengali

result = transliterate_hanzi_to_bengali("北京")  # বেইজিং
print(result)

# জাপানি প্রধানমন্ত্রীর নাম প্রতিবর্ণীকরণ
sys.path.insert(0, '../জাপানি-বাংলা অনুবাদ')
from japanese_transliteration import transliterate_to_bengali

result = transliterate_to_bengali("Kishida Fumio")  # কিশিদা ফুমিও
print(result)

# রুশ নাম সিরিলিক থেকে বাংলায়
sys.path.insert(0, '../রুশ-বাংলা অনুবাদ')
from cyrillic_transliteration import transliterate_cyrillic

result = transliterate_cyrillic("Путин")  # পুতিন
print(result)

# জার্মান নাম বাংলায়
sys.path.insert(0, '../জার্মান-বাংলা অনুবাদ')
from german_transliteration import transliterate_person_name

result = transliterate_person_name("Zimmermann")  # সিমারমান
print(result)
```
